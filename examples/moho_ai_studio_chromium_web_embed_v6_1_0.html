مقتطف الربط السريع بمنصات الذكاء الاصطناعي 

import {
  RealBrowserClient,
  createManagedBrowserAgent,
  createWindowsDesktopAgent,
  createLocalFolderBrain,
  createUnifiedAIToolExecutor,
  toGeminiFunctionDeclarations,
  toAnthropicTools,
  toOpenAITools,
  registerWithChromeWebMCP,
} from "chromium-web-embed";

const client = new RealBrowserClient({ label: "Moho AI" });
await client.connect();

const desktop = createWindowsDesktopAgent({ bridge: client, token: "YOUR_TOKEN" });
await desktop.connect();
await desktop.createVirtualDesktop(); // سطح مكتب افتراضي فارغ

const brain = createLocalFolderBrain();
await brain.pickFolder({ mode: "readwrite" });

const executor = createUnifiedAIToolExecutor({
  browserAgent: createManagedBrowserAgent(client),
  desktopAgent: desktop,
  folderBrain: brain,
});

// جاهز لـ Gemini / Claude / ChatGPT / MOHO + Chrome WebMCP
registerWithChromeWebMCP(executor);
const geminiTools = [{ functionDeclarations: toGeminiFunctionDeclarations() }];
const claudeTools = toAnthropicTools();
const openaiTools = toOpenAITools();
